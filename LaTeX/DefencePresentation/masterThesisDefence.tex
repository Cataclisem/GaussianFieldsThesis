	\begin{frame}{Hausdorff Dimension}
		
		\setcounter{chapCount}{1}
		\setcounter{theorems}{5}

		\vspace{3em}
		\begin{defi}
			Let $X$ be a metric space. If $K \subset X$ and $s \in [0, \infty)$ we define
			\vspace{-0.5em}
			\begin{equation}
				H^{s}_{\delta}(K) = \inf \cbr{\sum_{i =1}^{\infty} \diam(U_i)^s \ : \ \bigcup_{i = 1}^{\infty} U_i \supseteq K, \diam(U_i) < \delta}.
			\end{equation}
			
			\vspace{-0.5em}
			The $s$-dimensional \emph{Hausdorff measure} is then given by
			
			\vspace{-0.5em}
			\begin{equation}
				H^s(K) = \lim_{\delta \to 0} H^{s}_{\delta}(K).
			\end{equation}
		\end{defi}
		\pause 
		\setcounter{theorems}{7}
		\begin{defi}
			The \emph{Hausdorff dimension}, denoted $d_h(K)$ is given by
			
			\vspace{-1.5em}
			\begin{equation}
				d_h(K) = \inf\cbr{s \geq 0 : H^{s}(K) = 0} = \sup\cbr{s \geq 0 : H^{s}(K) = \infty}.
			\end{equation}
		\end{defi} 
	\end{frame}
	
	\begin{frame}{Hausdorff Dimension}
		
		\vspace{3em}
		\begin{theorems}
			Suppose the open set condition holds for the similarities $\cbr{\fracConst_i}_{1 \leq i \leq m}$ on $\ree^{n}$ with ratios $0 < c_i < 1$. If, for $K$ the following holds
			
			\vspace{-1em}
			\begin{equation}\label{hausDorffDimCondition}
				K = \bigcup_{i = 1}^{m} \fracConst_i(K)
			\end{equation}
			
			\vspace{-0.5em}
			then $d_h = s$ where $s$ is given by
			
			\vspace{-1em}
			\begin{equation}\label{hausDorffDimRatioSum}
				\sum_{i = 1}^{m} c_i^{s} = 1.
			\end{equation}
			
			\vspace{-0.5em}
			Moreover, for this value of $s$, $0 < H^s(K) < \infty$.
		\end{theorems}
		
		\pause

		\begin{description}
			\item[Open Set Condition: ] 	
				$ V \supset \bigcup_{i = 1}^{m} \fracConst_i(V), \quad \fracConst_i(V) \text{ disjoint}$
		\end{description}
	\end{frame}
	
	\begin{frame}{Spectral and Walk Dimension}
		\begin{description}
			\item<1->[Walk Dimension: ] $B_t$ Brownian motion on $K$
			\begin{equation}
				c_1 r^{d_w} \leq \e \left[\tau(x,r)\right] \leq c_2 r^{d_w},
			\end{equation}
			where $\tau(x,r) = \inf\cbr{t \geq 0 : B_t \notin B(x,r)}$.
			
			\vspace{1em}
			\item<2->[Spectral Dimension: ]  $d_s = \frac{2d_h}{d_w}$
		\end{description}
	\end{frame}

\begin{frame}{Main Theorem}
	\vspace{1em}
	\begin{lemma}\label{convergence}
		For all $f \in S(K)$ and $s \geq 0$ the following holds,
		\begin{equation}
			\frac{1}{a_m} \sum_{p \in V_m} f_m(p) (-\Delta_m)^{-2s}f_m(p) \to \int_{K}f(x)(-\Delta)^{-2s}f(x) \dd \mu(x)
		\end{equation}
		as $m \to \infty$.
	\end{lemma}

	\vspace{1em}
	\begin{mainTheorem}
		Let $s \geq 0$, then $X_{s}^{m}$ converges to $X_s$ in distribution in $S'(K)$ when $m \to \infty$.
	\end{mainTheorem}
\end{frame}

\begin{frame}
	\textcolor{blue}{Proof of Main Theorem: }
	
	
	\begin{itemize}
		\item<1-> Since $S(K)$ is a nuclear space.
		
		For all $f \in S(K)$,  
		$$\e \left[\exp \left(itX_{s}^{m}(f_m) \right)\right] \to \e \left[\exp\left(itX_{s}(f)\right)\right], \ \text{ as } m \to \infty$$
