	\section{Something}
	\subsection{Reformulated Lemma 2.4}
		\begin{frame}
		Let $x, y \in K$ then $\cbr{x} = x_{i_1, i_2, \ldots}$ and $\cbr{y} = y_{j_1, j_2, \ldots}$. If $x \underset{m}{\sim} y$ they must share an a $\alpha$ such $i_m = j_m$ for $m \leq \alpha$. 
		
		\setcounter{chapCount}{2}
		\setcounter{theorems}{3}
		\begin{lem}
			Let $u: V^{*} \to \ree$ be a harmonic function on the Vicsek set . If $v(x) = v(y) = 0$ and $y \in V_{m,x}$, then $v(\omega) = 0$ for all $\omega \in T_x$ where $T_x = \left( \bigcup_{\mathcal{I}_{i_1, \ldots, i_\alpha}} \fracConst_{i_1, i_2, \ldots}(K) \right)$.
		\end{lem}
		\begin{proof}
			Clear $v(\omega) = 0$ for $\omega \in T_x \bigcap V^{*}$ since $v$ is harmonic and depends on the values of its neighbours.
			
			Now let $\omega \in T_x \setminus V^{*}$ then $T_x \setminus V^{*}$ is dense in $T_x$ and the same arguments follow.
		\end{proof}
	\end{frame}
