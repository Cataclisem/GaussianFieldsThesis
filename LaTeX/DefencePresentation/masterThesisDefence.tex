	\begin{frame}{Hausdorff Dimension}
		
		\setcounter{chapCount}{1}
		\vspace{3em}
		\begin{theorems}
			Suppose the open set condition holds for the similarities $\cbr{\fracConst_i}_{1 \leq i \leq m}$ on $\ree^{n}$ with ratios $0 < c_i < 1$. If, for $K$ the following holds
			
			\vspace{-1em}
			\begin{equation}\label{hausDorffDimCondition}
				K = \bigcup_{i = 1}^{m} \fracConst_i(K)
			\end{equation}
			
			\vspace{-0.5em}
			then $d_h = s$ where $s$ is given by
			
			\vspace{-1em}
			\begin{equation}\label{hausDorffDimRatioSum}
				\sum_{i = 1}^{m} c_i^{s} = 1.
			\end{equation}
			
			\vspace{-0.5em}
			Moreover, for this value of $s$, $0 < H^s(K) < \infty$.
		\end{theorems}

	\section{Laplacians}
	\begin{frame}{Discrete Laplacian}
		
		\vspace{2em}
		
		\setcounter{theorems}{0}
		\setcounter{chapCount}{2}
		\begin{defi}\label{discreteLaplacian}
			Let $\ell(V_m)$ be the set of functions $\func{f}{V_m}{\ree}$. Then for any $f \in \ell(V_m)$  the \emph{discrete Laplacian} on $V_m$ is defined as
			\begin{equation}
				\Delta_m f(p) = 3^{m} \frac{1}{N_p}\sum_{q \in V_{m,p}} (f(q) - f(p)), \quad p \in V_m \setminus V_0
			\end{equation}
			where $$N_p = \begin{cases}
				5^{-m}, & \text{if } \ \# V_{m,p} = 4 \\
				2\cd 5^{-(m+1)}, & \text{if } \ \# V_{m,p} = 2\\
				5^{-(m+1)},  & \text{if } \ \# V_{m,p} = 1.
			\end{cases}$$
		\end{defi}
	\end{frame}
	
	\begin{frame}{Kigami Laplacian}
		\setcounter{theorems}{9}
		\vspace{3em}
		\begin{defi}
			Let $u \in \dom \mathcal{E}$ and let $f \in C(K)$, then $u \in \dom \Delta_{\mu}$ and $\Delta_{\mu} u =f$ if
			\begin{equation}
				\mathcal{E}(u,v) = \int_{K}fv \dd \mu, \quad \forall v \in \dom_{0}\mathcal{E}.
			\end{equation}
			We call $\Delta_{\mu}$ the \emph{Kigami Laplacian}.
		\end{defi}
		
		\pause
		
		\begin{description}
			\item<2->[Problem: ] Hard to find $\Delta_{\mu}u$
		\end{description}
		\begin{align}
			\dom_0 \mathcal{E} &= \cbr{u \in \ell(V^{*}) : \mathcal{E}(u, u) < \infty, \ u = 0,\text{ on } V_0 } \\ 
			&\phantom{a} \\
			\mathcal{E}(u, v) &= \lim_{m \to \infty} r^{-m}\sum_{x \sim y} \left(u(x) - u(y) \right)\left(v(x) - v(y) \right)
		\end{align}
		\end{frame}
		
		\begin{frame}{Kigami Laplacian}
			
		\setcounter{theorems}{15}
		\begin{theorems} \label{pointwiseFormula}
			Let $u \in \dom \Delta_{\mu}$. Then the pointwise formula
			\begin{equation}\label{pointwiseEquation}
				\Delta_{\mu} u(x) = \lim_{m \to \infty} r^{-m} \left(\int_{K} \psi_x^{(m)} \dd \mu \right)^{-1} \widetilde{\Delta}_m u(x)
			\end{equation}
			holds with the limit uniform across $V_* \setminus V_0$. Conversely, suppose $u$ is a continuous function and the right side of \eqref{pointwiseEquation} converges uniformly to a continuous function on $V_* \setminus V_0$. Then $u \in \dom \Delta_{\mu}$ and \eqref{pointwiseEquation} holds.
		\end{theorems}
	\end{frame}

	\begin{frame}{Semigroups and Heat Kernels}
		
		\vspace{3.5em}
		\setcounter{theorems}{24}
		\begin{theorems}
			If $A$ is a densely defined self adjoint operator on $\mathcal{H}$ then it is the generator of the strongly continuous self-adjoint contraction semigroup on $\mathcal{H}$ defined as $P_t = e^{-tA}$.
		\end{theorems}
		\pause
	\setcounter{theorems}{26}
	\begin{defi}\label{heatKernel}
		Let $(P_t)_{t \geq 0}$ be a strongly continuous self-adjoint contraction semigroup on $L^2(X, \mu)$. We say that the semigroup $(P_t)_{t \geq 0}$ \emph{admits a heat kernel} if there exists a measurable function $p\colon \ree_{> 0} \times X \times X \to \ree_{\geq 0}$ such that for every $t > 0$, a.e. we have
		
		\vspace{-0.5em}
		\begin{equation}
			P_tf(x) = \int_{K}p_t(x,y)f(y) \dd \mu(y)
		\end{equation}
		
	\vspace{-0.5em}
		with $x,y \in X, f \in L^{\infty}(X, \mu)$.
	\end{defi}
	\end{frame}
	
	\begin{frame}{Premliminary to Fractional Laplacian}
		
		\vspace{3.5em}
		\begin{itemize}
			\item<1-> Recall: $\Delta$ with $\dom \Delta$ is densely defined self-adjoint operator
			\item<2-> Generates strongly continuous self-adjoint contraction semigroup $P_t$
			\item<3-> $P_t$ admits heatkernel $p_t(x,y)$
		\end{itemize}
		
		\setcounter{theorems}{28}
		\begin{theorems}<4->\label{heatKernelWritten}
			There exists an orthonormal basis $\cbr{\phi_{i}}^{\infty}_{i = 1}$ of $L^2(K, \mu)$ with $\phi_j$ being eigenfunctions of $-\Delta$. If $\lambda_i$ is the eigenvalue belonging to $\phi_i$, then $\lambda_i >0$ and $\lim_{i \to \infty} \lambda_i = \infty$. Moreover
			
			\vspace{-0.5em}
			\begin{equation}
				p_t(x,y) = \sum_{j = 1}^{\infty} e^{-\lambda_j t} \phi_j(x)\phi_j(y),
			\end{equation}
			
			\vspace{-0.5em}
			where the convergence is uniform for all $t \in (0, \infty), \ x,y \in K \times K$.
		\end{theorems}
	\end{frame}
	
	\begin{frame}{Fractional Laplacian}
		
		\vspace{3em}

			\vspace{-0.5em}
			\begin{equation}
				(-\Delta)^{-s}f = \sum_{j = 1}^{\infty} \frac{1}{\lambda_{j}^{s}} \phi_j \int_{K} \phi_j(y)f(y) \dd \mu(y).
			\end{equation}
		\end{defi}

		\end{defi}
	\end{frame}

	\begin{frame}{Fractional Gaussian Fields}
		
		\vspace{2.5em}
		\begin{defi}\label{fracGaussianField}
			Let $s \geq 0$. A \emph{fractional Gaussian field} $X_s$ with parameter $s$ on $K$ is a centered Gaussian field, $\cbr{X_s(f): f \in S(K)}$, such that for any $f,g \in S(K)$
			\vspace{-1em}
			\begin{equation}
				\e \left[X_s(f) X_s(g)\right] = \int_{K} (-\Delta)^{-s}f (-\Delta)^{-s}g \dd \mu.
			\end{equation}
		\end{defi}
		\pause

\begin{frame}{Main Theorem}

	\begin{lem}\label{convergence}

		\begin{equation}
			\frac{1}{a_m} \sum_{p \in V_m} f_m(p) (-\Delta_m)^{-2s}f_m(p) \to \int_{K}f(x)(-\Delta)^{-2s}f(x) \dd \mu(x)
		\end{equation}
		as $m \to \infty$.
	\end{lem}

	\end{mainTheorem}

\begin{frame}

	\begin{itemize}

		For all $f \in S(K)$,  
		$$\e \left[\exp \left(itX_{s}^{m}(f_m) \right)\right] \to \e \left[\exp\left(itX_{s}(f)\right)\right], \ \text{ as } m \to \infty$$
		where $f_m = f \mid_{V_m}$. 
		
		\item<2-> Next: 
		\begin{multline}
			\e \left[\exp\left(itX_{s}^{m}(f_m)\right)\right] = \exp \left(-\frac{t^2}{2} \e \left[ \left(X_{s}^{m}(f_m) \right)^2\right]\right) \\
			= \exp \left(-\frac{t^2}{2a_m} \sum_{p \in V_m} f_m(p) (-\Delta_m)^{-2s} f_m(p)\right).

	\end{itemize}
